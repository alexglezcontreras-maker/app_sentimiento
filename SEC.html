  <script>

    const positiveKeywords = [
      "growth", "profit", "profits", "revenue", "strong", "improved",
      "success", "opportunity", "expansion", "positive", "innovation",
      "efficient", "increase", "gains", "record", "momentum", "leadership",
      "stable", "resilient", "advantage"
    ];

    const negativeKeywords = [
      "loss", "losses", "risk", "decline", "debt", "negative",
      "weakness", "lawsuit", "uncertain", "bankruptcy", "volatility",
      "drop", "decrease", "fraud", "litigation", "challenge", "concern",
      "downturn", "inflation", "default"
    ];

    function analyzeFile() {
      const input = document.getElementById('fileInput');
      const file = input.files[0];

      if (!file) {
        alert('Por favor sube un archivo primero.');
        return;
      }

      const reader = new FileReader();

      reader.onload = function(e) {
        const text = e.target.result.toLowerCase();

        let positiveCount = 0;
        let negativeCount = 0;

        const foundPositive = [];
        const foundNegative = [];

        positiveKeywords.forEach(word => {
          const matches = (text.match(new RegExp(word, 'g')) || []).length;
          if (matches > 0) {
            positiveCount += matches;
            foundPositive.push(`${word} (${matches})`);
          }
        });

        negativeKeywords.forEach(word => {
          const matches = (text.match(new RegExp(word, 'g')) || []).length;
          if (matches > 0) {
            negativeCount += matches;
            foundNegative.push(`${word} (${matches})`);
          }
        });

        const total = positiveCount + negativeCount;

        let score = 0;
        if (total > 0) {
          score = ((positiveCount - negativeCount) / total * 100).toFixed(2);
        }

        let sentiment = 'Neutral';
        let recommendation = 'Analizar más información';
        let color = 'text-yellow-400';

        if (score > 15) {
          sentiment = 'Positivo';
          recommendation = 'Podría valer la pena investigar esta inversión';
          color = 'text-emerald-400';
        }

        if (score < -15) {
          sentiment = 'Negativo';
          recommendation = 'Se recomienda precaución';
          color = 'text-red-400';
        }

        document.getElementById('results').classList.remove('hidden');

        const sentimentLabel = document.getElementById('sentimentLabel');
        sentimentLabel.textContent = sentiment;
        sentimentLabel.className = `text-2xl font-bold ${color}`;

        document.getElementById('score').textContent = `${score}%`;
        document.getElementById('recommendation').textContent = recommendation;

        document.getElementById('summary').innerHTML = `
          El archivo SEC contiene <strong>${positiveCount}</strong> menciones positivas y
          <strong>${negativeCount}</strong> menciones negativas.
          El análisis general del documento se clasifica como
          <strong>${sentiment}</strong>.
        `;

        document.getElementById('positiveWords').innerHTML =
          foundPositive.length > 0
            ? foundPositive.map(w => `<li>• ${w}</li>`).join('')
            : '<li>No se detectaron palabras positivas.</li>';

        document.getElementById('negativeWords').innerHTML =
          foundNegative.length > 0
            ? foundNegative.map(w => `<li>• ${w}</li>`).join('')
            : '<li>No se detectaron palabras negativas.</li>';
      };

      reader.readAsText(file);
    }

    function clearResults() {
      document.getElementById('results').classList.add('hidden');
      document.getElementById('fileInput').value = '';
    }

  </script>

Cómo usarlo
